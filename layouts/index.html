{{ define "main" }} {{- $s := .Site -}} {{- $sections := $s.Params.mainSections | default (slice "posts") -}} {{- $posts := where $s.RegularPages "Type" "in" $sections -}} {{- $n := int ($s.Params.postsOnHome | default 6) -}} {{- /* postsOnHomeMax renders further posts that a "Show more" button reveals in place — the middle ground between a short home page and the full archive, without a second request. Leave it unset (or at or below postsOnHome) and nothing changes. The extras are rendered visible and JavaScript hides them, not the other way round, so a reader without JavaScript gets more posts rather than a button that does nothing. */ -}} {{- $max := int ($s.Params.postsOnHomeMax | default $n) -}} {{- if lt $max $n }}{{ $max = $n }}{{ end -}} {{- $recent := first $max $posts -}} {{- /* Where "Archive · N" points. Prefer a real /archives/ page. */ -}} {{- $archive := "" -}} {{- with $s.GetPage "/archives" }}{{ $archive = .RelPermalink }}{{ else }} {{- with $s.GetPage (printf "/%s" (index $sections 0)) }}{{ $archive = .RelPermalink }}{{ end -}} {{- end -}}

{{- with $s.Params.masthead -}} {{ . | safeHTML }} {{- else -}} {{ $s.Title }}. {{- end -}}
