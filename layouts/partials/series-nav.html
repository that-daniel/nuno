{{- /* Series navigation: where this post sits in its series, and every other part, linked. Ordering is chronological unless *every* part declares `seriesOrder`, in which case that wins. Half-ordered is worse than unordered — one post with an explicit number and the rest without produces a list that looks deliberate and is not — so it is all or nothing, and the fallback is the order they were published in. A post in more than one series gets a block per series. That is unusual but not wrong, and silently showing only the first would be. Renders nothing when the post is in no series, or in a series of one: "Part 1 of 1" is noise. Takes the page as its argument. */ -}} {{- $page := . -}} {{- range $page.GetTerms "series" }} {{- $parts := .Pages -}} {{- $numbered := where $parts "Params.seriesOrder" "!=" nil -}} {{- if and (gt (len $parts) 0) (eq (len $numbered) (len $parts)) -}} {{- $parts = sort $parts "Params.seriesOrder" -}} {{- else -}} {{- $parts = sort $parts "Date" -}} {{- end -}} {{- if gt (len $parts) 1 }} {{- $index := 0 -}} {{- range $i, $p := $parts }} {{- if eq $p.RelPermalink $page.RelPermalink }}{{ $index = add $i 1 }}{{ end -}} {{- end }}
{{ i18n "seriesLabel" }} · {{ .LinkTitle }} {{- with $index }} {{ i18n "seriesPart" (dict "n" . "total" (len $parts)) }} {{- end }}
    {{- range $parts }} {{- if eq .RelPermalink $page.RelPermalink }}
  1. {{ .LinkTitle }}
    2. {{- else }}
  2. {{ .LinkTitle }}
    3. {{- end }} {{- end }}
{{- end }} {{- end }}
